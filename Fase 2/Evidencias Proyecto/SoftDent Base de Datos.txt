CREATE TABLE public.pacientes (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  numero_ficha bigint GENERATED ALWAYS AS IDENTITY NOT NULL UNIQUE,
  rut text NOT NULL UNIQUE,
  extranjero boolean NOT NULL DEFAULT false,
  nombres text NOT NULL,
  apellidos text NOT NULL,
  nombre_social text,
  fecha_nacimiento date NOT NULL CHECK (fecha_nacimiento <= CURRENT_DATE AND fecha_nacimiento >= '1900-01-01'::date),
  sexo text CHECK (sexo IS NULL OR (sexo = ANY (ARRAY['FEMENINO'::text, 'MASCULINO'::text, 'INTERSEXUAL'::text, 'NO_INFORMA'::text]))),
  genero text,
  telefono text,
  telefono_fijo text,
  email text CHECK (email IS NULL OR email ~* '^[^@\s]+@[^@\s]+\.[^@\s]+$'::text),
  direccion text,
  comuna text,
  ciudad text,
  prevision text NOT NULL DEFAULT 'Particular'::text,
  actividad_profesion text,
  empleador text,
  referencia text,
  observaciones text,
  apoderado_nombre text,
  apoderado_rut text,
  apoderado_telefono text,
  titular_id uuid,
  activo boolean NOT NULL DEFAULT true,
  motivo_deshabilitado text,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  modificado_por uuid,
  CONSTRAINT pacientes_pkey PRIMARY KEY (id),
  CONSTRAINT pacientes_titular_id_fkey FOREIGN KEY (titular_id) REFERENCES public.pacientes(id),
  CONSTRAINT pacientes_modificado_por_fkey FOREIGN KEY (modificado_por) REFERENCES public.perfiles(id)
);
CREATE TABLE public.antecedentes (

  contenido text NOT NULL CHECK (length(TRIM(BOTH FROM contenido)) > 0),
  vigente boolean NOT NULL DEFAULT true,
  profesional_id uuid,

  fecha_baja timestamp with time zone,
  motivo_baja text,
  baja_por uuid,
  CONSTRAINT antecedentes_pkey PRIMARY KEY (id),
  CONSTRAINT antecedentes_paciente_id_fkey FOREIGN KEY (paciente_id) REFERENCES public.pacientes(id),
  CONSTRAINT antecedentes_profesional_id_fkey FOREIGN KEY (profesional_id) REFERENCES public.perfiles(id),
  CONSTRAINT antecedentes_baja_por_fkey FOREIGN KEY (baja_por) REFERENCES public.perfiles(id)
);
CREATE TABLE public.historial_paciente (

CREATE TABLE public.archivos_paciente (

CREATE TABLE public.perfiles (
  id uuid NOT NULL,
  nombre_completo text NOT NULL CHECK (length(TRIM(BOTH FROM nombre_completo)) > 0),
  rut text UNIQUE CHECK (rut IS NULL OR rut ~ '^[0-9]{7,8}-[0-9K]$'::text),
  email text NOT NULL UNIQUE,
  rol text NOT NULL CHECK (rol = ANY (ARRAY['ADMINISTRADOR'::text, 'DENTISTA'::text, 'ASISTENTE'::text, 'RECEPCIONISTA'::text])),
  especialidad text,
  registro_profesional text,
  activo boolean NOT NULL DEFAULT true,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  porcentaje_liquidacion numeric CHECK (porcentaje_liquidacion IS NULL OR porcentaje_liquidacion >= 0::numeric AND porcentaje_liquidacion <= 100::numeric),
  intentos_fallidos integer DEFAULT 0,
  bloqueado_hasta timestamp with time zone,
  firma_svg text,
  preferencias jsonb,
  telefono text,
  agenda_online boolean NOT NULL DEFAULT true,
  CONSTRAINT perfiles_pkey PRIMARY KEY (id),
  CONSTRAINT perfiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);

CREATE TABLE public.prestaciones (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  codigo_interno text NOT NULL UNIQUE CHECK (codigo_interno ~ '^[A-Z0-9-]{2,20}$'::text),
  codigo_fonasa text CHECK (codigo_fonasa IS NULL OR codigo_fonasa ~ '^[0-9]{7}$'::text),
  nombre text NOT NULL CHECK (length(TRIM(BOTH FROM nombre)) >= 3),
  descripcion text,
  categoria_id uuid NOT NULL,
  especialidad_id uuid,
  duracion_min integer NOT NULL DEFAULT 30 CHECK (duracion_min >= 5 AND duracion_min <= 480 AND (duracion_min % 5) = 0),
  activo boolean NOT NULL DEFAULT true,
  creado_por uuid,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  icono text,
  laboratorio_id uuid,
  CONSTRAINT prestaciones_pkey PRIMARY KEY (id),
  CONSTRAINT prestaciones_categoria_id_fkey FOREIGN KEY (categoria_id) REFERENCES public.categorias_prestaciones(id),
  CONSTRAINT prestaciones_especialidad_id_fkey FOREIGN KEY (especialidad_id) REFERENCES public.especialidades(id),
  CONSTRAINT prestaciones_creado_por_fkey FOREIGN KEY (creado_por) REFERENCES public.perfiles(id),
  CONSTRAINT prestaciones_laboratorio_id_fkey FOREIGN KEY (laboratorio_id) REFERENCES public.laboratorios(id)
);
